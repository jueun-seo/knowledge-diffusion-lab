### 2. HTTP 에러 객체에 상태 코드(error.status) 담아 던지기 `code: frontend/src/api/api.js`
```
export const SERVER_URL = 'http://localhost:8000'

export async function api(path, options) {
  const res = await fetch(`${SERVER_URL}${path}`, options)

  if (!res.ok) {
    const data = await res.json()
    const error = new Error(data.detail)
    error.status = res.status
    throw error
  }
  if (res.status === 204) return null
  return await res.json()
}
```
- 개념: JWT 및 쿠키 인증, 예외 처리
- 메모: 실패 응답 시 `error.status`에 HTTP 상태 코드를 저장해 `throw`. `401` 인증 오류를 구분해 재발급 로직으로 넘기기 위함.

### 3. authApi에서 Bearer 토큰 자동 부착 및 401 시 refresh 재발급·재요청 구현 `code: frontend/src/hooks/useAuth.jsx`
```
async function authApi(path, options) {
  try {
    return await api(path, {
      ...options,
      headers: {
        ...options?.headers,
        Authorization: `Bearer ${token}`,
      },
    })
  } catch (e) {
    if (e.status !== 401) throw e

    const data = await api('/api/auth/refresh', {
      method: 'POST',
      credentials: 'include',
    })
    setToken(data.access_token)
    return await api(path, {
      ...options,
      headers: {
        ...options?.headers,
        Authorization: `Bearer ${data.access_token}`,
      },
    })
  }
}
```
- 개념: JWT 및 쿠키 인증, 예외 처리
- 메모: `401` 발생 시 쿠키 포함 `/api/auth/refresh`로 새 액세스 토큰을 받아 저장하고 `data.access_token`으로 재요청. 토큰 만료로 인한 기능 멈춤 방지.

### 7. NewPost에서 사진 업로드(FormData -> /api/uploads) 후 게시물 등록(/api/posts) 및 홈 이동 `code: frontend/src/routes/NewPost.jsx`
```
import { useState } from 'react'
import { useNavigate } from 'react-router'
import { useAuth } from '../hooks/useAuth.jsx'

function NewPost() {
  const [content, setContent] = useState('')
  const [file, setFile] = useState(null)
  const [error, setError] = useState('')
  const [sending, setSending] = useState(false)
  const { authApi } = useAuth()
  const navigate = useNavigate()

  async function handleSubmit(e) {
    e.preventDefault()
    setSending(true)
    try {
      let image_url = null
      if (file) {
        const formData = new FormData()
        formData.append('photo', file)
        const data = await authApi('/api/uploads', { method: 'POST', body: formData })
        image_url = data.image_url
      }

      await authApi('/api/posts', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ content, image_url }),
      })
      navigate('/')
    } catch (e) {
      setError(e.message)
      setSending(false)
    }
  }

  return (
    <div className="card p-3">
      <h2 className="h5 mb-3">새 게시물</h2>
      <form className="d-grid gap-2" onSubmit={handleSubmit}>
        <input className="form-control" type="file" accept="image/*" onChange={e => setFile(e.target.files[0])} />
        {file && (
          <div className="ratio ratio-1x1">
            <img src={URL.createObjectURL(file)} className="object-fit-cover rounded" />
          </div>
        )}
        <textarea className="form-control" rows={3} value={content} onChange={e => setContent(e.target.value)} placeholder="문구 입력…" required />
        <button className="btn btn-primary" disabled={sending}>{sending ? '보내는중...' : '공유하기'}</button>
      </form>
      {error && <p className="text-danger mt-3 mb-0">{error}</p>}
    </div>
  )
}

export default NewPost
```
- 개념: React 상태 관리, JWT 및 쿠키 인증
- 메모: 파일이 있으면 `FormData`로 `/api/uploads`에서 `image_url`을 먼저 받은 뒤 `content`와 함께 `/api/posts`에 저장하고 `/`로 이동. 중복 전송 방지를 위해 `sending` 상태 사용.

### 11. photos.py의 save_photo에서 퍼센트 좌표를 픽셀로 변환해 크롭 후 저장 `code: backend/photos.py`
```
import uuid
from pathlib import Path
from fastapi import UploadFile, HTTPException
from PIL import Image, ImageOps, UnidentifiedImageError

Path('media').mkdir(exist_ok=True)

def save_photo(photo: UploadFile, x: float = 0, y: float = 0, width: float = 100, height: float = 100):
  try:
    image = Image.open(photo.file)
  except UnidentifiedImageError:
    raise HTTPException(status_code=400, detail='이미지 파일만 올릴 수 있습니다.')

  image = ImageOps.exif_transpose(image)

  w, h = image.size
  left = round(w * x / 100)
  top = round(h * y / 100)
  right = round(w * (x + width) / 100)
  bottom = round(h * (y + height) / 100)
  image = image.crop((left, top, right, bottom))

  image = ImageOps.fit(image, (1080, 1080))
  image = image.convert('RGB')

  name = f'{uuid.uuid4().hex}.jpg'
  image.save(f'media/{name}')

  return f'/media/{name}'
```
- 개념: 예외 처리
- 메모: 회전 보정 후 백분율 좌표를 픽셀 `(left, top, right, bottom)`으로 계산해 `crop`하고 `1080x1080`·`RGB`로 저장. 지정한 영역대로 잘린 이미지를 반환하기 위함.

### 13. CommentItem에서 내 댓글 삭제 버튼 표시, DELETE 요청 및 인라인 수정 모드 구현 `code: frontend/src/components/CommentItem.jsx`
```
import { useState } from 'react'
import { useAuth } from '../hooks/useAuth.jsx'

function CommentItem({ id, user, content, onDelete, onUpdate }) {
  const [editing, setEditing] = useState(false)
  const [text, setText] = useState(content)
  const auth = useAuth()
  const mine = auth.user && user && user.id === auth.user.id

  function handleCancel() {
    setText(content)
    setEditing(false)
  }

  async function handleSave(e) {
    e.preventDefault()
    const updated = await auth.authApi(`/api/comments/${id}`, {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ content: text }),
    })
    onUpdate(updated)
    setEditing(false)
  }

  async function handleDelete() {
    await auth.authApi(`/api/comments/${id}`, {
      method: 'DELETE',
    })
    onDelete(id)
  }

  if (editing) {
    return (
      <li className="list-group-item">
        <form className="input-group input-group-sm" onSubmit={handleSave}>
          <input
            className="form-control"
            type="text"
            value={text}
            onChange={e => setText(e.target.value)}
            required
          />
          <button type="submit" className="btn btn-primary">저장</button>
          <button type="button" className="btn btn-outline-secondary" onClick={handleCancel}>취소</button>
        </form>
      </li>
    )
  }

  return (
    <li className="list-group-item d-flex justify-content-between">
      <span><b>{user.username}</b> {content}</span>
      {mine && (
        <span className="d-flex gap-2">
          <button type="button" className="btn btn-sm btn-link p-0" onClick={() => setEditing(true)}>수정</button>
          <button type="button" className="btn btn-sm btn-link text-danger p-0" onClick={handleDelete}>삭제</button>
        </span>
      )}
    </li>
  )
}

export default CommentItem
```
- 개념: React 상태 관리, JWT 및 쿠키 인증
- 메모: 작성자 일치(`mine`) 시 수정·삭제 버튼을 보여 주고, `PATCH`/`DELETE` API 요청 후 각각 `onUpdate`, `onDelete`로 부모 상태를 갱신. 댓글 목록 자리에서 바로 수정하고 지우기 위함.

### 14. PostDetail에서 handleCreate·handleDelete·handleUpdate로 comments 상태 관리 `code: frontend/src/routes/PostDetail.jsx`
```
function handleCreate(created) {
  setComments([...comments, created])
}

function handleDelete(commentId) {
  setComments(comments.filter(c => c.id !== commentId))
}

function handleUpdate(updated) {
  setComments(comments.map(c => {

```

## 개념 설명
### 예외 처리 `Backend`
예상치 못한 오류나 잘못된 입력·인증 실패 상황을 잡아내고 적절한 에러 객체나 HTTP 응답으로 처리하는 방식이며, 예를 들어 이미지가 아닌 파일을 열 때 `UnidentifiedImageError`를 잡아 `400` 상태 코드를 반환함. 프론트엔드에서는 `error.status`에 HTTP 상태 코드를 담아 `401` 여부를 판별하고, 백엔드 `save_photo`에서는 회전 보정 뒤 크롭 좌표를 픽셀로 환산해 이미지를 자른 후 저장함.
HTTP 상태 코드: 요청이 성공했는지(`200`, `201`, `204`) 인증이 안 됐는지(`401`) 잘못된 요청인지(`400`)를 나타내는 숫자 신호.
백분율 크롭 좌표(`percentCrop`): 화면 크기가 달라져도 원본 이미지의 가로·세로 비율에 맞춰 `left, top, right, bottom` 픽셀 좌표로 환산해 자르기 위한 비율 값.

### React 상태 관리 `Frontend`
컴포넌트가 가진 데이터(`state`)가 바뀔 때 화면을 자동으로 다시 그리도록 관리하는 방식이며, 예를 들어 `comments` 상태에 새 댓글을 넣으면 댓글 목록이 즉시 갱신됨. 상위 컴포넌트가 상태를 들고 상태 변경 함수를 만들며, 하위 컴포넌트가 이벤트와 API 요청을 마친 뒤 그 함수를 호출해 상위 데이터를 바꾸는 패턴이 반복됨. 입력 폼과 상태를 묶는 제어 컴포넌트 방식이나 `URL.createObjectURL(file)`로 선택한 사진 미리보기를 띄우는 기능도 상태 변화에 따라 화면을 갱신하는 역할을 함.
삼항 연산자: 조건이 참일 때와 거짓일 때 값을 `조건 ? 참 : 거짓` 한 줄로 선택하는 문법.
조건부 렌더링: 상태나 조건(`file && ...`, `editing`)에 따라 보여 줄 화면 요소를 다르게 그리는 방식.
제어 컴포넌트: 입력창의 값(`value`)을 `useState` 상태와 연결하고 `onChange`로 상태를 갱신해 입력과 상태를 일치시키는 방식.

### JWT 및 쿠키 인증 `Backend`
짧은 유효기간의 액세스 토큰과 긴 유효기간의 리프레시 토큰을 조합해 로그인 상태를 유지하고 만료 시 자동으로 갱신하는 인증 구조이며, 예를 들어 액세스 토큰이 1분 뒤 만료되어 `401`이 뜨면 쿠키의 리프레시 토큰으로 새 액세스 토큰을 받아 원래 요청을 다시 보냄. 요청마다 `Authorization: Bearer ...` 헤더를 붙이고 기존 옵션 헤더(`...options?.headers`)를 함께 펼쳐 `Content-Type` 같은 설정이 지워지지 않게 유지함. 파일 전송에는 `FormData`를 사용하고 JSON 본문 전송에는 `'Content-Type': 'application/json'`과 `JSON.stringify`를 사용함.
리프레시 토큰 재발급: 만료된 액세스 토큰으로 `401`이 발생할 때 쿠키를 포함(`credentials: 'include'`)해 `/api/auth/refresh`로 새 액세스 토큰을 발급받는 과정.
멀티파트 폼 데이터(`FormData`): 이미지 파일이나 크롭 좌표 같은 값을 서버로 전송할 때 사용하는 폼 데이터 규격.
